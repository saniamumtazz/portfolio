# LLM-powered code assistant
from fastapi import FastAPI
from openai import OpenAI

app = FastAPI()
client = OpenAI()

@app.post("/explain")
def explain(code: str):
    prompt = "Explain and debug:"
    return client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user",
                   "content": prompt + code}]
    )
